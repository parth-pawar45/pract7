3. Write jQuery to show the use of click(), hover(), on(), 
trigger(),off() events. 
<!DOCTYPE html>
<html>
<head>
    <title>Basic jQuery</title>

    <script src="jquery-3.6.1.min.js"></script>

    <script>
        $(document).ready(function () {

            // hover()
            $("#b1").hover(function () {
                $("#msg").text("Hello Romal");
            });

            // on()
            $("p").on("click", function () {
                $(this).css("background", "yellow");
            });

            // off()
            $("#b2").click(function () {
                $("p").off("click");
            });

            // on()
            $("#b3").on("click", function () {
                $("#t1").hide();
            });

            // select()
            $("input").select(function () {
                $("input").after("Text Marked!");
            });

            // trigger()
            $("#b4").click(function () {
                $("input").trigger("select");
            });

        });
    </script>
</head>

<body>

    <button id="b1">Hover</button>
    <p id="msg"></p>

    <p>Welcome to the Mulund College</p>
    <p>Romal: 239720</p>

    <button id="b2">Off</button>

    <p id="t1">Hello Romal</p>

    <button id="b3">On</button>

    <br><br>

    <input type="text" value="Romal 239720">

    <button id="b4">Trigger</button>
</body></html>
